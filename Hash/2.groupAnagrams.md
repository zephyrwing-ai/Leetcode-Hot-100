class Solution(object):
    def sort_string(self,str):
        a=sorted([x for x in str])
        return "".join(a)
    def groupAnagrams(self, strs):
        """
        :type strs: List[str]
        :rtype: List[List[str]]
        """
        grouped_string={}
        for i in strs:
            a=self.sort_string(i)
            if a in grouped_string:
                grouped_string[a].append(i)
            else:
                grouped_string[a]=[i]
        return list(grouped_string.values())
