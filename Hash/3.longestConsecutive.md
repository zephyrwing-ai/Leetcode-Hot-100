class Solution:
    def longestConsecutive(self, nums: List[int]) -> int:
        num=set(nums) #这里要使用set将nums转化为使用哈希结构，不然的话会导致后面if判断语句中的时间复杂度过高
        maxlength=0
        for i in num:
            if i-1 not in num:
                current_number=i
                current_streak=1
                while current_number+1 in num:
                    current_number+=1
                    current_streak+=1
                maxlength=max(maxlength,current_streak)
        return maxlength
