# 3Sum

```python
class Solution(object):
    def threeSum(self, nums):
        res=[]
        num=sorted(nums) #进行排序后，就是从小到大排列，那么就是说当second+third大于target的时候，就只需要将third往右移动就行了
        n=len(num)
        for first in range(n):
            if first>0 and num[first]==num[first-1]: #对first大于0进行判断的原因是当first大于0的时候，也就是进入了可能第二次甚至第三次的循环，这时候就有可能会出现重复的值了，所以才加了这一个判断，后面找second以及third都是一样的逻辑
                continue
            target=-num[first]  #这里直接计算出target用于后面判断second+third的值是不是大于target，好进行移动相应值
            third=n-1
            for second in range(first+1,n):
                if second>first+1 and num[second]==num[second-1]:
                    continue
                while second<third and num[second]+num[third]>target:
                        third-=1
                if second == third:
                    break
                if num[second]+num[third]==target:
                    res.append([num[first],num[second],num[third]])
        return res
```
